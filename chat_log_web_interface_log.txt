  Why PostgreSQL Over Neo4j/Pure Vector DB

  1. Speed: SQL queries are 10-100x faster for structured queries
  2. Reporting: SQL excels at aggregations, joins, and complex reports
  3. Maturity: Battle-tested with excellent tooling
  4. Hybrid: PostgreSQL + pgvector gives you both SQL and vector search
  5. Simplicity: Single database for everything

  Proposed Schema

  -- Core tables matching the taxonomy
  CREATE TABLE sessions (
      session_id UUID PRIMARY KEY,
      cwd TEXT,
      git_branch TEXT,
      created_at TIMESTAMPTZ,
      metadata JSONB
  );

  CREATE TABLE nodes (
      uuid UUID PRIMARY KEY,
      session_id UUID REFERENCES sessions,
      parent_uuid UUID,
      node_type TEXT, -- 'user', 'assistant', 'system', 'summary'
      timestamp TIMESTAMPTZ,
      content_hash TEXT,
      is_sidechain BOOLEAN DEFAULT FALSE,
      raw_json JSONB -- Keep original for flexibility
  );

  CREATE TABLE tool_calls (
      id SERIAL PRIMARY KEY,
      node_uuid UUID REFERENCES nodes,
      tool_use_id TEXT UNIQUE,
      tool_name TEXT,
      input_json JSONB,
      timestamp TIMESTAMPTZ
  );

  CREATE TABLE tool_results (
      id SERIAL PRIMARY KEY,
      tool_use_id TEXT,
      node_uuid UUID REFERENCES nodes,
      is_error BOOLEAN,
      content_text TEXT,
      content_json JSONB,
      timestamp TIMESTAMPTZ
  );

  CREATE TABLE file_operations (
      id SERIAL PRIMARY KEY,
      tool_call_id TEXT,
      operation TEXT, -- 'read', 'write', 'edit'
      file_path TEXT,
      line_start INT,
      line_end INT,
      content_before TEXT,
      content_after TEXT
  );

  -- Indexes for fast queries
  CREATE INDEX idx_nodes_session ON nodes(session_id);
  CREATE INDEX idx_nodes_timestamp ON nodes(timestamp);
  CREATE INDEX idx_tool_calls_name ON tool_calls(tool_name);
  CREATE INDEX idx_file_ops_path ON file_operations(file_path);
  CREATE INDEX idx_nodes_type ON nodes(node_type);

  -- Optional: Vector embeddings for semantic search
  CREATE EXTENSION IF NOT EXISTS vector;

  CREATE TABLE embeddings (
      id SERIAL PRIMARY KEY,
      node_uuid UUID REFERENCES nodes,
      embedding vector(1536), -- OpenAI ada-002 dimension
      model TEXT DEFAULT 'ada-002'
  );

  CREATE INDEX ON embeddings USING ivfflat (embedding vector_cosine_ops);
