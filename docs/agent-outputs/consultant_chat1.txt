  """
  Bidirectional integration between agent execution and knowledge base.

  import subprocess
  import json
  from typing import Dict, Any, Optional
  from pathlib import Path

  class KnowledgeFeedbackLoop:
      """Connects agent execution to persistent learning"""

      def __init__(self):
          self.knowledge_cache = {}
          self.pattern_buffer = []

      async def on_agent_start(self, job_id: str, task: str, agent_role: str):
          """Query knowledge before agent starts"""
          # Pull relevant knowledge for this task type
          result = subprocess.run([
              "mcp", "call", "mcp__knowledge__knowledge_smart_query",
              "--query", f"{agent_role} performing {task}",
              "--limit", "3"
          ], capture_output=True, text=True)

          if result.returncode == 0:
              knowledge = json.loads(result.stdout)
              self.knowledge_cache[job_id] = knowledge
              return knowledge.get("results", [])
          return []

      async def on_agent_error(self, job_id: str, error: str, context: Dict):
          """Store error patterns for future avoidance"""
          pattern = {
              "error": error,
              "agent_role": context.get("agent_role"),
              "task": context.get("task"),
              "phase": context.get("phase"),
              "timestamp": context.get("timestamp")
          }

          # Immediate storage for critical errors
          if "permission denied" in error.lower() or "api key" in error.lower():
              subprocess.run([
                  "mcp", "call", "mcp__knowledge__knowledge_store",
                  "--content", json.dumps(pattern),
                  "--knowledge_type", "ERROR_SOLUTION",
                  "--tags", f"auto-captured,agent:{job_id},critical"
              ])
          else:
              # Buffer non-critical for batch storage
              self.pattern_buffer.append(pattern)
              if len(self.pattern_buffer) >= 10:
                  await self.flush_patterns()

      async def on_agent_success(self, job_id: str, result: Dict):
          """Store success patterns for replication"""
          # Update confidence of applied knowledge
          if job_id in self.knowledge_cache:
              for knowledge_item in self.knowledge_cache[job_id]:
                  subprocess.run([
                      "mcp", "call", "mcp__knowledge__knowledge_update",
                      "--knowledge_id", knowledge_item.get("id"),
                      "--update_type", "success"
                  ])

          # Store new success pattern
          pattern = {
              "task": result.get("task"),
              "approach": result.get("tools_used"),
              "duration": result.get("duration"),
              "outputs": result.get("files_created")
          }

          subprocess.run([
              "mcp", "call", "mcp__knowledge__knowledge_store",
              "--content", json.dumps(pattern),
              "--knowledge_type", "WORKFLOW",
              "--tags", f"success-pattern,agent:{job_id}"
          ])

  class OptimizedTokenPoolManager(TokenPoolJobManager):
      """Enhanced manager for 15-agent coordination"""

      def __init__(self, config):
          super().__init__(config)
          self.MAX_CONCURRENT = 15
          self.token_health = {i: {"status": "healthy", "failures": 0}
                               for i in range(1, 16)}
          self.task_queue = asyncio.Queue()
          self.results_collector = {}

      async def execute_phase_with_15_agents(
          self,
          phase_config: Dict,
          timeout: int = 3600
      ) -> Dict[str, Any]:
          """Execute a phase using all 15 agents optimally"""

          start_time = asyncio.get_event_loop().time()
          phase_name = phase_config["name"]

          # Start knowledge feedback loop
          feedback_loop = KnowledgeFeedbackLoop()

          # Create 15 worker coroutines
          workers = []
          for token_id in range(1, 16):
              worker = self.agent_worker(
                  token_id=token_id,
                  phase_name=phase_name,
                  feedback_loop=feedback_loop
              )
              workers.append(asyncio.create_task(worker))

          # Distribute tasks across workers
          all_tasks = self.generate_phase_tasks(phase_config)
          for task in all_tasks:
              await self.task_queue.put(task)

          # Add sentinel values to stop workers
          for _ in range(15):
              await self.task_queue.put(None)

          # Wait for all workers with timeout
          try:
              results = await asyncio.wait_for(
                  asyncio.gather(*workers),
                  timeout=timeout
              )
          except asyncio.TimeoutError:
              self.logger.error(f"Phase {phase_name} timed out after {timeout}s")
              # Cancel remaining tasks
              for worker in workers:
                  worker.cancel()

          # Collect and analyze results
          phase_results = {
              "phase": phase_name,
              "duration": asyncio.get_event_loop().time() - start_time,
              "agents_used": 15,
              "tasks_completed": len(self.results_collector),
              "success_rate": self.calculate_success_rate(),
              "token_utilization": self.calculate_token_utilization()
          }

          return phase_results

      async def agent_worker(
          self,
          token_id: int,
          phase_name: str,
          feedback_loop: KnowledgeFeedbackLoop
      ):
          """Worker coroutine for a single agent/token"""

          while True:
              task = await self.task_queue.get()
              if task is None:  # Sentinel
                  break

              job_id = f"{phase_name}_{token_id}_{task['id']}"

              try:
                  # Query knowledge before starting
                  knowledge = await feedback_loop.on_agent_start(
                      job_id, task['description'], task['agent_role']
                  )

                  # Execute task with token
                  result = await self.execute_single_task(
                      task, token_id, knowledge
                  )

                  # Record success
                  await feedback_loop.on_agent_success(job_id, result)
                  self.results_collector[job_id] = result

              except Exception as e:
                  # Record failure
                  await feedback_loop.on_agent_error(
                      job_id, str(e), {"task": task, "token": token_id}
                  )

                  # Mark token as unhealthy if too many failures
                  self.token_health[token_id]["failures"] += 1
                  if self.token_health[token_id]["failures"] >= 3:
                      self.token_health[token_id]["status"] = "unhealthy"
                      self.logger.warning(f"Token {token_id} marked unhealthy")

                  # Requeue task for another agent
                  await self.task_queue.put(task)
