  # 1) Add a pluggable “Response Validator” chain (opt-in)

  Purpose: turn runtime observations (e.g., “NOT\_JSON”) into **actionable gates** that (a) fail the task, (b) propagate failure to dependents, and
   (c) log structured reasons you can mine to auto-improve prompts/policies.

  ### A) New file: `apps/agent-orchestrator/src/validators/base.py`

  ````python
  # lightweight plugin-style validators for agent stdout
  from typing import Any, Dict, Tuple, Optional
  import json, os, re

  class ValidationError(Exception):
      def __init__(self, code: str, detail: str, raw: str):
          super().__init__(f"{code}: {detail}")
          self.code, self.detail, self.raw = code, detail, raw

  class BaseValidator:
      name = "base"
      def validate(self, stdout_text: str, task: Any) -> Dict[str, Any]:
          return {"status": "pass", "payload": {"raw_text": stdout_text}}

  class StripFences(BaseValidator):
      name = "strip_fences"
      def validate(self, stdout_text: str, task: Any) -> Dict[str, Any]:
          if os.getenv("UNIFIED_STRIP_FENCES","0") != "1":
              return {"status":"pass","payload":{"raw_text":stdout_text}}
          m = re.search(r"```(?:json)?\s*(.*?)\s*```", stdout_text, re.S|re.I)
          if m:
              stdout_text = m.group(1).strip()
          return {"status":"pass","payload":{"raw_text":stdout_text}}

  class JsonParse(BaseValidator):
      name = "json_parse"
      def validate(self, stdout_text: str, task: Any) -> Dict[str, Any]:
          try:
              obj = json.loads(stdout_text)
              return {"status":"pass","payload":obj}
          except Exception as e:
              if os.getenv("UNIFIED_REQUIRE_JSON","0") == "1":
                  raise ValidationError("json_parse_failed", str(e)[:200], stdout_text)
              return {"status":"pass","payload":{"status":"completed","raw_text":stdout_text}}

  # Optional, plug in JSON Schema when desired
  try:
      import jsonschema
      class JsonSchema(BaseValidator):
          name = "json_schema"
          def validate(self, stdout_text: str, task: Any) -> Dict[str, Any]:
              if os.getenv("UNIFIED_VALIDATE_SCHEMA","0") != "1":
                  return {"status":"pass","payload":json.loads(stdout_text)}
              schema = {
                  "type":"object",
                  "properties":{"status":{"const":"completed"}, "task_id":{"type":"string"}},
                  "required":["status"],
                  "additionalProperties": True
              }
              obj = json.loads(stdout_text)
              jsonschema.validate(obj, schema)
              if "task_id" in obj and obj["task_id"] != getattr(task, "task_id", obj["task_id"]):
                  raise ValidationError("schema_task_id_mismatch","task_id differs from task.task_id", stdout_text)
              return {"status":"pass","payload":obj}
  except Exception:
      JsonSchema = None

  def default_chain():
      chain = [StripFences(), JsonParse()]
      if JsonSchema: chain.append(JsonSchema())
      return chain
  ````

  # 2) Turn your validation into a **Self-Improvement Suite**

  Create a declarative suite that both **runs** and **asserts** expectations, then writes improvement hints based on observed failures.

  ### A) `validation_suite.json`

  ```json
  {
    "suite_id": "unified_e2e_core",
    "expectations": {
      "A_fast_1": "completed",
      "A_fast_2": "completed",
      "A_slow_1": "completed",
      "A_slow_2": "completed",
      "B_join":   "completed",
      "C_bad_json": "failed",
      "D_after_bad": "failed",
      "E_deadlock":  "failed"
    },
    "hints": {
      "json_parse_failed": "Enable UNIFIED_STRIP_FENCES=1 or tighten prompts to emit bare JSON.",
      "schema_task_id_mismatch": "Add explicit task_id echo to prompt or relax schema.",
      "dependencies_not_met": "Check upstream task failures; consider retry/backoff policy."
    }
  }
  ```

  ### B) `tools/validate_and_improve.py`

  ```python
  #!/usr/bin/env python3
  import json, sys, os, glob, time
  from pathlib import Path

  OUT = Path(os.environ.get("ORCH_OUT", "./_run_e2e_unified"))
  SUITE = Path("validation_suite.json")

  def load_statuses():
      tasks = {}
      for p in glob.glob(str(OUT / "tasks/*.json")):
          j = json.load(open(p))
          tasks[j["task_id"]] = j.get("status") or j.get("state","unknown")
      return tasks

  def main():
      suite = json.load(open(SUITE))
      exp = suite["expectations"]
      statuses = load_statuses()

      failures = {}
      for tid, want in exp.items():
          got = statuses.get(tid, "missing")
          if got != want:
              failures[tid] = {"want": want, "got": got}

      report = {
          "suite_id": suite["suite_id"],
          "timestamp": time.time(),
          "pass": len(failures) == 0,
          "failures": failures,
          "improvement_hints": []
      }

      if failures:
          # map observed deltas to hints
          hints = suite.get("hints", {})
          # pull error_details if present
          for tid in failures:
              p = OUT / f"tasks/{tid}.json"
              if p.exists():
                  j = json.load(open(p))
                  err = j.get("error_details","")
                  for code, hint in hints.items():
                      if code in err:
                          report["improvement_hints"].append({"task": tid, "code": code, "hint": hint})

      outp = OUT / "self_improve_report.json"
      json.dump(report, open(outp,"w"), indent=2)
      print(f"Wrote {outp}")
      if not report["pass"]:
          sys.exit(2)

  if __name__ == "__main__":
      main()
  ```

  **How this “self-improves”:** every run creates `self_improve_report.json` that tells you **exactly** what to tweak (env flags, prompt style,
  schema) to recover to green. Feed those hints into either:

  * a **prompt patcher** (next section), or
  * a **policy file** (below) that turns knobs automatically on next run.

  # 3) Auto-tune parallelism & retries (safe hill-climb)

  Add a tiny controller that adjusts concurrency between waves to maximize throughput while keeping failures low.

  ### `apps/agent-orchestrator/src/auto_tune.py`

  ```python
  class AdaptiveParallelism:
      def __init__(self, initial:int, lo:int=2, hi:int=64):
          self.cur = initial; self.lo = lo; self.hi = hi

      def update(self, success_rate: float, avg_latency: float):
          if success_rate > 0.98 and avg_latency < 3.0 and self.cur < self.hi:
              self.cur += 1
          elif success_rate < 0.9 and self.cur > self.lo:
              self.cur -= 1

      def value(self) -> int:
          return self.cur
  ```

  **Wire it in** (non-breaking): on each scheduling cycle, compute `success_rate/latency` from your metrics and call `update()`. This makes the
  orchestrator **learn a better `max_parallel`** over time—no restarts.
